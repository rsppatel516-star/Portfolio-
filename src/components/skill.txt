const getTechHexColor = (icon) => {
  const isLight = typeof document !== 'undefined' && (
    document.documentElement.getAttribute('data-theme') === 'light' ||
    document.documentElement.classList.contains('light')
  );
  const raw = techHexColors[icon] || '#6366f1';
  if (isLight && (raw.toLowerCase() === '#ffffff' || raw.toLowerCase() === '#fff')) {
    return '#4f46e5';
  }
  return raw;
};

function TechCard({ tech, onInspect, isSelected }) {
  const [coords, setCoords] = useState({ x: 0, y: 0 });
  const [isHovered, setIsHovered] = useState(false);
  const cardRef = useRef(null);

  const handleMouseMove = (e) => {
    if (!cardRef.current) return;
    const rect = cardRef.current.getBoundingClientRect();
    setCoords({
      x: e.clientX - rect.left,
      y: e.clientY - rect.top
    });
  };

  const hex = getTechHexColor(tech.icon);

  return (
    <motion.div
      ref={cardRef}
      variants={cardVariant}
      onMouseMove={handleMouseMove}
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}

      whileHover={{ y: -6, transition: { duration: 0.25 } }}

function TechInspectorModal({ tech, onClose }) {
  // Find real portfolio projects matching this technology
  const matchingProjects = useMemo(() => {
    if (!tech) return [];
    const t = tech.name.toLowerCase().trim();
    const cleanT = t.replace(/\.js$/, '').replace(/[^a-z0-9]/g, '');

    return projects.filter((proj) => {
      const skillsMatch = proj.skillsUsed?.some((s) => {
        const cleanS = s.toLowerCase().replace(/\.js$/, '').replace(/[^a-z0-9]/g, '');
        return cleanS.includes(cleanT) || cleanT.includes(cleanS);
      });
      const tagsMatch = proj.tags?.some((tg) => {
        const cleanTg = tg.toLowerCase().replace(/[^a-z0-9]/g, '');
        return cleanTg.includes(cleanT) || cleanT.includes(cleanTg);
      });
      return skillsMatch || tagsMatch;
    });
  }, [tech]);

  // Handle ESC key
  useEffect(() => {
    const handleKeyDown = (e) => {
      if (e.key === 'Escape') onClose();
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [onClose]);

  if (!tech) return null;

  const hex = getTechHexColor(tech.icon);

  return (
    <motion.div
      initial={{ opacity: 0 }}
      animate={{ opacity: 1 }}
      exit={{ opacity: 0 }}
      onClick={onClose}
      className="fixed inset-0 z-50 flex items-center justify-center p-4 sm:p-6 bg-black/60 dark:bg-black/80 backdrop-blur-md"

  const [searchQuery, setSearchQuery] = useState('');
  const [selectedTag, setSelectedTag] = useState('ALL');
  const [sortBy, setSortBy] = useState('mastery'); // 'mastery' | 'projects' | 'az'
  const [viewMode, setViewMode] = useState('grid'); // 'grid' | 'compact'
  const [inspectTech, setInspectTech] = useState(null);

  // Dynamic quick-filter tags (Non-siloed smart filters)
  const filterChips = [
    { id: 'ALL', label: 'All Stack', count: allTechnologies.length, icon: Layers },
    { id: 'CORE', label: '★ Core Production (90%+)', count: allTechnologies.filter(t => t.proficiency >= 90).length, icon: Star },
    { id: 'DAILY', label: '✦ Daily Drivers', count: allTechnologies.filter(t => t.badge === 'Daily Driver' || t.highlight).length, icon: Flame },
    { id: 'APPLE', label: '🍎 Apple & iOS', count: allTechnologies.filter(t => t.domain === 'Native iOS').length, icon: Smartphone },
    { id: 'WEB', label: '🌐 Web & Cloud', count: allTechnologies.filter(t => t.domain === 'Web Ecosystem' || t.domain === 'Backend & Cloud').length, icon: Globe },
    { id: 'AI', label: '🤖 AI Tools', count: allTechnologies.filter(t => t.domain === 'AI & Toolchain').length, icon: Bot }
  ];

  // Filtered & sorted array of skills
  const filteredTechs = useMemo(() => {
    let result = [...allTechnologies];

    // Filter chip application
    if (selectedTag === 'CORE') {
      result = result.filter(t => t.proficiency >= 90);
    } else if (selectedTag === 'DAILY') {
      result = result.filter(t => t.badge === 'Daily Driver' || t.highlight);
    } else if (selectedTag === 'APPLE') {
      result = result.filter(t => t.domain === 'Native iOS');
    } else if (selectedTag === 'WEB') {
      result = result.filter(t => t.domain === 'Web Ecosystem' || t.domain === 'Backend & Cloud');
    } else if (selectedTag === 'AI') {
      result = result.filter(t => t.domain === 'AI & Toolchain');
    }

    // Search query application
    if (searchQuery.trim() !== '') {
      const q = searchQuery.toLowerCase().trim();
      result = result.filter(t =>
        t.name.toLowerCase().includes(q) ||
        t.role.toLowerCase().includes(q) ||
        t.domain.toLowerCase().includes(q) ||
        t.description.toLowerCase().includes(q)
      );
    }

    // Sort order
